name: Apply content triage hard delete

on:
  workflow_dispatch:
  push:
    branches: ["main"]
    paths:
      - ".github/workflows/apply-content-triage-2026-08-13.yml"
      - "supabase/migrations/20260813090000_hard_delete_triaged_projects.sql"
      - "scripts/apply-content-triage-2026-08-13.mjs"
      - "scripts/purge-triaged-project-assets.mjs"

permissions:
  contents: read

jobs:
  apply-production:
    runs-on: ubuntu-latest
    environment: production
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Pack apply bundle
        run: |
          mkdir -p .triage-npm
          npm install --prefix .triage-npm ali-oss@6.23.0 --omit=dev --no-fund --no-audit
          tar -czf triage-bundle.tgz \
            scripts/apply-content-triage-2026-08-13.mjs \
            scripts/db-push.mjs \
            scripts/purge-triaged-project-assets.mjs \
            scripts/lib/content-triage-2026-08-13.mjs \
            lib/utils/oss-client.mjs \
            supabase/migrations/20260813090000_hard_delete_triaged_projects.sql \
            -C .triage-npm node_modules

      - name: Copy bundle to server
        uses: appleboy/scp-action@v0.1.7
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SSH_PRIVATE_KEY }}
          port: ${{ secrets.SERVER_PORT || 22 }}
          source: triage-bundle.tgz
          target: /tmp/

      - name: Apply OSS purge then db:push
        uses: appleboy/ssh-action@v1
        env:
          ALIYUN_OSS_ACCESS_KEY_ID: ${{ secrets.ALIYUN_OSS_ACCESS_KEY_ID }}
          ALIYUN_OSS_ACCESS_KEY_SECRET: ${{ secrets.ALIYUN_OSS_ACCESS_KEY_SECRET }}
          ALIYUN_OSS_BUCKET: ${{ secrets.ALIYUN_OSS_BUCKET }}
          ALIYUN_OSS_REGION: ${{ secrets.ALIYUN_OSS_REGION }}
          ALIYUN_OSS_ENDPOINT: ${{ secrets.ALIYUN_OSS_ENDPOINT }}
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SSH_PRIVATE_KEY }}
          port: ${{ secrets.SERVER_PORT || 22 }}
          command_timeout: 30m
          envs: ALIYUN_OSS_ACCESS_KEY_ID,ALIYUN_OSS_ACCESS_KEY_SECRET,ALIYUN_OSS_BUCKET,ALIYUN_OSS_REGION,ALIYUN_OSS_ENDPOINT
          script: |
            set -euo pipefail
            echo 'env names in .env.production matching OSS/ALIYUN:'
            grep -E '^[A-Z0-9_]*(OSS|ALIYUN)[A-Z0-9_]*=' /opt/steam-app/.env.production | cut -d= -f1 || true
            rm -rf /tmp/triage-apply
            mkdir -p /tmp/triage-apply
            tar -xzf /tmp/triage-bundle.tgz -C /tmp/triage-apply
            rm -f /tmp/triage-bundle.tgz
            extra=()
            for name in ALIYUN_OSS_ACCESS_KEY_ID ALIYUN_OSS_ACCESS_KEY_SECRET ALIYUN_OSS_BUCKET ALIYUN_OSS_REGION ALIYUN_OSS_ENDPOINT; do
              eval "val=\${$name:-}"
              if [ -n "$val" ]; then extra+=(-e "$name"); fi
            done
            docker run --rm \
              --user root \
              --env-file /opt/steam-app/.env.production \
              "${extra[@]}" \
              -v /tmp/triage-apply:/work \
              -w /work \
              steam-app:latest \
              node scripts/apply-content-triage-2026-08-13.mjs --execute
            rm -rf /tmp/triage-apply
